# Task Performance Monitor

自动监控任务执行时间，智能决定是否使用 subagent。

## 原理

1. **记录执行历史** - 每次任务记录耗时
2. **自动分类** - 根据历史数据判断任务类型
3. **智能路由** - 耗时任务自动走 subagent

## 配置文件

`~/.openclaw/workspace/.task-monitor.json`:
```json
{
  "thresholds": {
    "slow_task_ms": 30000,
    "many_api_calls": 10
  },
  "history": {
    "gmail-auto-processor": {
      "avg_time_ms": 45000,
      "api_calls": 50,
      "last_run": "2026-02-09T14:00:00Z",
      "recommended_mode": "subagent"
    }
  },
  "auto_rules": [
    {
      "pattern": "gmail|email|mail",
      "mode": "subagent",
      "reason": "API调用多，容易超时"
    },
    {
      "pattern": "batch|bulk|批量",
      "mode": "subagent",
      "reason": "处理量大"
    },
    {
      "pattern": "crawl|spider|爬虫",
      "mode": "subagent",
      "reason": "网络IO耗时"
    }
  ]
}
```

## 使用方式

### 方式一：包装函数（推荐）

```javascript
// 在任何 skill 中使用
const { runTask } = require('../task-monitor');

// 自动决定是否走 subagent
await runTask('处理邮件', async () => {
  // 你的耗时任务
  await processEmails();
});
```

### 方式二：手动标记

```javascript
const { shouldUseSubagent } = require('../task-monitor');

if (shouldUseSubagent('gmail-processor')) {
  // 走 subagent
  spawnSubagent('处理邮件...');
} else {
  // 直接执行
  processEmails();
}
```

## 自动规则

| 任务特征 | 检测方式 | 自动路由 |
|---------|---------|---------|
| 执行历史 >30s | 记录上次耗时 | subagent |
| API 调用 >10次 | 预估调用量 | subagent |
| 关键词匹配 | gmail/batch/crawl | subagent |
| 文件处理 >100个 | 文件数量 | subagent |
| 网络请求 >20个 | HTTP 调用 | subagent |

## 学习机制

1. **首次执行** - 直接运行，记录耗时
2. **第二次执行** - 如果上次超时，提示用户是否用 subagent
3. **第三次执行** - 自动使用 subagent，无需询问

## 实现代码

```javascript
// task-monitor.js
const fs = require('fs');
const path = require('path');

const DB_PATH = path.join(__dirname, '.task-monitor.json');

class TaskMonitor {
  constructor() {
    this.db = this.loadDB();
  }

  loadDB() {
    if (fs.existsSync(DB_PATH)) {
      return JSON.parse(fs.readFileSync(DB_PATH, 'utf8'));
    }
    return { thresholds: { slow_task_ms: 30000 }, history: {}, auto_rules: [] };
  }

  saveDB() {
    fs.writeFileSync(DB_PATH, JSON.stringify(this.db, null, 2));
  }

  // 判断是否应该使用 subagent
  shouldUseSubagent(taskName, options = {}) {
    // 1. 检查历史记录
    const history = this.db.history[taskName];
    if (history && history.avg_time_ms > this.db.thresholds.slow_task_ms) {
      return { useSubagent: true, reason: `历史平均耗时 ${history.avg_time_ms}ms` };
    }

    // 2. 检查自动规则
    for (const rule of this.db.auto_rules) {
      if (taskName.match(new RegExp(rule.pattern, 'i'))) {
        return { useSubagent: true, reason: rule.reason };
      }
    }

    // 3. 检查预估 API 调用量
    if (options.apiCalls && options.apiCalls > this.db.thresholds.many_api_calls) {
      return { useSubagent: true, reason: `预估 API 调用 ${options.apiCalls} 次` };
    }

    return { useSubagent: false };
  }

  // 记录任务执行
  recordTask(taskName, durationMs, metadata = {}) {
    if (!this.db.history[taskName]) {
      this.db.history[taskName] = { runs: [] };
    }
    
    const history = this.db.history[taskName];
    history.runs.push({
      time: new Date().toISOString(),
      duration_ms: durationMs,
      ...metadata
    });
    
    // 只保留最近10次
    history.runs = history.runs.slice(-10);
    
    // 计算平均耗时
    history.avg_time_ms = history.runs.reduce((a, b) => a + b.duration_ms, 0) / history.runs.length;
    history.last_run = new Date().toISOString();
    
    // 自动推荐模式
    if (history.avg_time_ms > this.db.thresholds.slow_task_ms) {
      history.recommended_mode = 'subagent';
    }
    
    this.saveDB();
  }
}

module.exports = new TaskMonitor();
```

## 集成到 Skill

```javascript
// 在 skill 入口使用
const monitor = require('../task-monitor');

async function main() {
  const taskName = 'gmail-auto-processor';
  const recommendation = monitor.shouldUseSubagent(taskName, { apiCalls: 50 });
  
  if (recommendation.useSubagent) {
    console.log(`⏱️  检测到耗时任务 (${recommendation.reason})，使用 subagent 执行...`);
    // 走 subagent
  } else {
    const start = Date.now();
    // 直接执行
    const result = await process();
    // 记录耗时
    monitor.recordTask(taskName, Date.now() - start);
  }
}
```

## 优势

1. **零配置** - 自动学习，无需手动设置
2. **渐进式** - 不会第一次就误判
3. **可解释** - 每次决策都有原因
4. **可覆盖** - 用户可以强制指定模式
